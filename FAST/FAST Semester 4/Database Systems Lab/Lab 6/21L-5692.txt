SELECT SUBSTRING(Text, CHARINDEX('#', Text), CHARINDEX(' ', Text + ' ', CHARINDEX('#', Text)) - CHARINDEX('#', Text)) AS Hashtag, 
       UserName, 
       COUNT(*) AS Count 
FROM Tweets 
GROUP BY SUBSTRING(Text, CHARINDEX('#', Text), CHARINDEX(' ', Text + ' ', CHARINDEX('#', Text)) - CHARINDEX('#', Text)), UserName;

SELECT Users.Country, COUNT(*) AS Count 
FROM Users 
JOIN Tweets ON Users.UserName = Tweets.UserName

SELECT UserName 
FROM (
  SELECT UserName, COUNT(*) AS Count 
  FROM Tweets 
  GROUP BY UserName
) AS tweet_counts 
WHERE Count > (
SELECT AVG(Count) 
FROM (
SELECT COUNT(*) AS Count 
FROM Tweets 
GROUP BY UserName) 
AS tweet_counts);
