CREATE TRIGGER [dbo].[PreventCustomerDeletion]
ON [dbo].[Customers]
INSTEAD OF DELETE
AS
BEGIN
    IF EXISTS (
        SELECT 1 FROM [dbo].[Order] WHERE [CustomerNo] IN (SELECT [CustomerNo] FROM DELETED)
    )
    BEGIN
        PRINT 'Cannot delete customer who has made an order.'
    END
    ELSE
    BEGIN
        DELETE [dbo].[Customers] WHERE [CustomerNo] IN (SELECT [CustomerNo] FROM DELETED)
    END
END

CREATE TRIGGER [dbo].[CheckQuantityInStore]
ON [dbo].[OrderDetails]
INSTEAD OF INSERT
AS
BEGIN
    DECLARE @ItemNo INT
    DECLARE @Quantity INT
    SELECT @ItemNo = [ItemNo], @Quantity = [Quantity] FROM INSERTED
    
    DECLARE @QuantityInStore INT
    SELECT @QuantityInStore = [Quantity in Store] FROM [dbo].[Items] WHERE [ItemNo] = @ItemNo
    
    IF @Quantity <= @QuantityInStore
    BEGIN
        INSERT [dbo].[OrderDetails] ([OrderNo], [ItemNo], [Quantity]) VALUES ((SELECT [OrderNo] FROM INSERTED), @ItemNo, @Quantity)
    END
    ELSE
    BEGIN
        PRINT 'The required quantity is not available'
    END
END

CREATE TRIGGER [dbo].[CheckOrderDate]
ON [dbo].[Order]
INSTEAD OF INSERT
AS
BEGIN
  SET NOCOUNT ON
  DECLARE @currentDate DATE
  SET @currentDate = GETDATE()
  INSERT INTO [dbo].[Order] ([OrderNo], [CustomerNo], [Date], [Total_Items_Ordered])
  SELECT [OrderNo], [CustomerNo], CASE WHEN [Date] = @currentDate THEN [Date] ELSE @currentDate END, [Total_Items_Ordered] FROM inserted
END

CREATE TRIGGER [dbo].[PreventNullCustomerInsert]
ON [dbo].[Customers]
AFTER INSERT
AS
BEGIN
  IF EXISTS (SELECT * FROM [inserted] WHERE [Name] IS NULL OR [City] IS NULL OR [Phone] IS NULL)
  BEGIN
    RAISERROR ('Cannot insert customer with null fields.', 16, 1)
    ROLLBACK TRANSACTION
  END
END

CREATE PROCEDURE [dbo].[CustomerSignup]
  @CustomerNo varchar(2),
  @Name varchar(30),
  @City varchar(3),
  @Phone varchar(11)
AS
BEGIN
  IF (@CustomerNo IS NULL OR @Name IS NULL OR @City IS NULL OR @Phone IS NULL)
  BEGIN
    RAISERROR ('Cannot insert customer with null fields.', 16, 1)
    RETURN
  END
  
  INSERT INTO [dbo].[Customers] ([CustomerNo], [Name], [City], [Phone])
  VALUES (@CustomerNo, @Name, @City, @Phone)
END
