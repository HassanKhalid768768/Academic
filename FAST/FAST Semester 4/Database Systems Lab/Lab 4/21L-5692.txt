 -- create a table named booking

create table booking(
hotelno varchar(10),
guestno numeric(5),
datefrom date,
dateto date,
roomno numeric(5),
primary key (hotelno, guestno, datefrom),
foreign key (roomno, hotelno) REFERENCES room(roomno, hotelno),
foreign key (guestno) REFERENCES guest(guestno)
)

 -- Question 9

SELECT DISTINCT *
FROM hotel 
JOIN room r1 ON hotel.hotelno = r1.hotelno AND r1.type = 'double'
JOIN room r2 ON hotel.hotelno = r2.hotelno AND r2.type = 'family';

 -- Question 10

SELECT DISTINCT *
FROM hotel
JOIN room r1 ON hotel.hotelno = r1.hotelno AND r1.type = 'single'
JOIN room r2 ON hotel.hotelno = r2.hotelno AND r2.type = 'double'
LEFT JOIN room r3 ON hotel.hotelno = r3.hotelno AND r3.type = 'family'
WHERE r3.roomno IS NULL;
