create table [User](
[userId] int primary key,
[name] varchar(20) not null,
[phoneNum] varchar(15) not null,
[city] varchar(20) not null
)

create table CardType(
[cardTypeID] int primary key,
[name] varchar(15),
[description] varchar(40) null
)

create Table [Card](
cardNum Varchar(20) primary key,
cardTypeID int foreign key references  CardType([cardTypeID]),
PIN varchar(4) not null,
[expireDate] date not null,
balance float not null
)

Create table UserCard(
userID int foreign key references [User]([userId]),
cardNum varchar(20) foreign key references [Card](cardNum),
primary key(cardNum)
)

create table [Transaction](
transId int primary key,
transDate date not null,
cardNum varchar(20) foreign key references [Card](cardNum),
amount int not null
)

CREATE PROCEDURE Login
    @cardNum VARCHAR(16),
    @PIN VARCHAR(4),
    @Status BIT OUTPUT
AS
BEGIN
    SET @Status = 0;
    IF EXISTS (SELECT  FROM Card WHERE cardNum = @cardNum AND PIN = @PIN)
    BEGIN
        SET @Status = 1;
    END;
END;

CREATE PROCEDURE UpdatePIN
    @CardNum VARCHAR(16),
    @OldPIN VARCHAR(4),
    @NewPIN VARCHAR(4)
AS
BEGIN
    IF LEN(@NewPIN) = 4
    BEGIN
        UPDATE Card
        SET PIN = @NewPIN
        WHERE cardNum = @CardNum AND PIN = @OldPIN;

        IF @@ROWCOUNT  0
        BEGIN
            PRINT 'Updated PIN';
        END
        ELSE
        BEGIN
            PRINT 'Error';
        END;
    END
    ELSE
    BEGIN
        PRINT 'Error';
    END;
END;

CREATE PROCEDURE WithDraw
    @cardNum VARCHAR(16),
    @PIN VARCHAR(4),
    @amount DECIMAL(10, 2)
AS
BEGIN
    DECLARE @status BIT
    DECLARE @balance DECIMAL(10, 2)
    DECLARE @transID INT
    
    -- Check if card number and PIN are valid
    EXEC Login @cardNum, @PIN, @status OUTPUT
    
    IF @status = 1
    BEGIN
        -- Get the user's balance
        SELECT @balance = balance
        FROM Card
        WHERE cardNum = @cardNum
        
        -- Check if the user has sufficient balance
        IF @balance = @amount
        BEGIN
            -- Get the next transaction ID
            SELECT @transID = ISNULL(MAX(transId), 0) + 1
            FROM [Transaction]
            
            -- Insert transaction details
            INSERT INTO [Transaction] (transID, cardNum, transDate, amount)
            VALUES (@transID, @cardNum, GETDATE(), @amount)
            
            -- Update the user's balance
            UPDATE Card
            SET balance = balance - @amount
            WHERE cardNum = @cardNum
        END
        ELSE
        BEGIN
            -- Insert failed transaction details
            SELECT @transID = ISNULL(MAX(transID), 0) + 1
            FROM [Transaction]
            
            INSERT INTO [Transaction] (transID, cardNum, transDate, amount)
            VALUES (@transID, @cardNum, GETDATE(), @amount)
        END
    END
    ELSE
    BEGIN
        -- Insert failed transaction details
        SELECT @transID = ISNULL(MAX(transID), 0) + 1
        FROM [Transaction]
        
        INSERT INTO [Transaction] (transID, cardNum, transDate, amount)
        VALUES (@transID, @cardNum, GETDATE(), @amount)
    END
END
