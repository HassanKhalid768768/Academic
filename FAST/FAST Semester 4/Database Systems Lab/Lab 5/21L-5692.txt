create table booking(
hotelno varchar(10),
guestno numeric(5),
datefrom datetime,
dateto datetime,
roomno numeric(5),
primary key (hotelno, guestno, datefrom),
foreign key (roomno, hotelno) REFERENCES room(roomno, hotelno),
foreign key (guestno) REFERENCES guest(guestno)
)

-- Question 1

SELECT guestname, guestaddress
FROM guest
WHERE guestaddress LIKE '%London%'
ORDER BY guestname ASC;

-- Question 2

SELECT hotelname, COUNT(*) AS num_rooms
FROM hotel
JOIN room ON hotel.hotelno = room.hotelno
GROUP BY hotelname;

-- Question 3

SELECT hotelname, AVG(price) AS avg_price
FROM hotel
JOIN room ON hotel.hotelno = room.hotelno
WHERE city = 'London'
GROUP BY hotelname;

-- Question 4

SELECT type, MAX(price) AS max_price
FROM room
GROUP BY type;

-- Question 5

SELECT hotel.hotelname, hotel.city, COUNT(DISTINCT type) AS num_room_types
FROM hotel
JOIN room ON hotel.hotelno = room.hotelno
GROUP BY hotel.hotelname, hotel.city;

-- Question 6

SELECT DISTINCT hotel.hotelname, hotel.city
FROM hotel
JOIN room ON hotel.hotelno = room.hotelno
JOIN booking b ON room.hotelno = b.hotelno AND room.roomno = b.roomno
JOIN guest g ON b.guestno = g.guestno
WHERE g.guestaddress LIKE '%London%';

-- Question 7

SELECT hotel.hotelname, hotel.city, COUNT(*) AS num_reservations
FROM hotel
JOIN room ON hotel.hotelno = room.hotelno
JOIN booking ON room.hotelno = booking.hotelno AND room.roomno = booking.roomno
GROUP BY hotel.hotelname, hotel.city
ORDER BY num_reservations DESC;

-- Question 8

SELECT guest.guestname
FROM guest
JOIN booking ON guest.guestno = booking.guestno
WHERE dateto IS NULL;

-- Question 9

SELECT hotel.hotelname, room.roomno, booking.guestno
FROM hotel
JOIN room ON hotel.hotelno = room.hotelno
JOIN booking ON room.hotelno = booking.hotelno AND room.roomno = booking.roomno
WHERE YEAR(datefrom) IN (2001, 2002);

-- Question 10

SELECT hotel.hotelname, hotel.city
FROM hotel
LEFT JOIN room ON hotel.hotelno = room.hotelno
WHERE room.roomno IS NULL;

-- Question 11

SELECT hotelname, hotel.hotelno, roomno, price
FROM hotel
LEFT JOIN room ON hotel.hotelno = room.hotelno AND type = 'family'

-- Question 12
